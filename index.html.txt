  <script>
    const depInput = document.getElementById('deposits');
    const topInput = document.getElementById('topups');
    const result = document.getElementById('result');

    function format(num) {
      return '$' + num.toFixed(2);
    }

    function calc() {
      const dep = parseFloat(depInput.value) || 0;
      const top = parseFloat(topInput.value) || 0;

      // Комісії
      const depComm = dep * 0.20;
      const topComm = top * 0.10;
      const earned = depComm + topComm;

      // Розподіл комісії
      const model = earned * 0.40;
      const chatter = earned * 0.60;

      // Общак (залишок)
      const obshak = (dep * 0.80) + (top * 0.90);

      document.getElementById('depComm').textContent = format(depComm);
      document.getElementById('topComm').textContent = format(topComm);
      document.getElementById('earned').textContent = format(earned);
      document.getElementById('model').textContent = format(model);
      document.getElementById('chatter').textContent = format(chatter);
      document.getElementById('obshak').textContent = format(obshak);

      if (dep > 0 || top > 0) {
        result.classList.add('show');
      } else {
        result.classList.remove('show');
      }
    }

    depInput.addEventListener('input', calc);
    topInput.addEventListener('input', calc);
  </script>
